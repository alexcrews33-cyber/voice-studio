from fastapi import FastAPI, UploadFile, File, Form
from fastapi.middleware.cors import CORSMiddleware
from fastapi.responses import FileResponse
from rvc_python.infer import RVCInference
import shutil
import os

app = FastAPI()

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

UPLOAD_DIR = "temp_audio"
os.makedirs(UPLOAD_DIR, exist_ok=True)

# Initialize RVC Inference
rvc = RVCInference(device="cuda:0") # Change to "cpu" if no GPU

@app.post("/convert")
async def convert_voice(
    file: UploadFile = File(...),
    model_name: str = Form("morgan.pth"),
    f0_up_key: int = Form(0),
    index_rate: float = Form(0.75)
):
    input_path = os.path.join(UPLOAD_DIR, f"input_{file.filename}")
    output_path = os.path.join(UPLOAD_DIR, f"output_{file.filename}")

    with open(input_path, "wb") as buffer:
        shutil.copyfileobj(file.file, buffer)

    # Execute RVC Inference
    rvc.load_model(f"models/{model_name}")
    rvc.infer_file(
        input_path=input_path,
        output_path=output_path,
        f0_up_key=f0_up_key,
        index_rate=index_rate
    )

    return FileResponse(output_path, media_type="audio/wav")

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8000)
Save as main.py & run python main.py
